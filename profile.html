Sustainability profile

Which kinds of sustainable plastics an actor works on, compared with another actor or with the world

Provisional results. The corpora are being validated: every number may change.

Choose the documents (publications or patents), an actor and an actor to compare it with. The first chart shows the share of the actor’s documents in each of the eight categories; the second its specialisation index, its share divided by the world’s share (above 1: the actor works on that category more than the world average); the third how its profile changed over time.

actors  = FileAttachment("data/profile_actors.csv").csv({typed: true})
cats    = FileAttachment("data/categories.csv").csv()
// periods and trend are read as text and the counts made numbers by hand:
// the automatic reading would take labels such as "2015-19" for dates
toNumbers = rows => rows.map(d => {
  const o = {...d}
  for (const k of ["n", ...cats.map(c => c.category)]) o[k] = +d[k]
  return o
})
periods = toNumbers(await FileAttachment("data/profile_periods.csv").csv())
trend   = toNumbers(await FileAttachment("data/profile_trend.csv").csv())

// R writes TRUE/FALSE: read them as yes/no
isTrue = v => v === true || v === "TRUE"
// the colours of the eight categories (the same as the graphs of the paper)
catColours = ["#2a78d6", "#eb6834", "#1baf7a", "#eda100", "#e87ba4", "#008300", "#6250d6", "#e34948"]
fmt = n => n.toLocaleString("en")
pct = x => (100 * x).toFixed(1) + "%"
viewof corpus = Inputs.radio(["Publications", "Patents"], {label: "Documents", value: "Publications"})
viewof actorType = Inputs.select(["Research organisation", "Company", "Country", "Area"],
                                 {label: "Type of actor", value: "Research organisation"})
viewof piedmontOnly = Inputs.toggle({label: "Only Piedmont", value: false})
choices = actors
  .filter(d => d.corpus === corpus && d.actor_type === actorType && (!piedmontOnly || isTrue(d.piedmont)))
  .sort((a, b) => b.docs - a.docs)

// UniTo is chosen at the start, when it is in the list
viewof actor = Inputs.select(choices, {
  label: "Actor",
  format: d => `${d.actor} (${fmt(d.docs)})`,
  value: choices.find(d => d.actor === "UniTo" || d.actor === "UNIVERSITA DEGLI STUDI DI TORINO")
})
others = actors.filter(d => d.corpus === corpus)
  .sort((a, b) => (a.actor_type === "Area" ? 0 : 1) - (b.actor_type === "Area" ? 0 : 1) || b.docs - a.docs)
viewof other = Inputs.select(others, {
  label: "Compare with",
  format: d => `${d.actor} (${fmt(d.docs)})`,
  value: others.find(d => d.actor_id === "A:World")
})
viewof period = Inputs.radio(["All years", "1980-1999", "2000-2014", "2015-2025"],
                             {label: "Period", value: "All years"})

Numbers in brackets: documents in 1980–2025. Patents: countries and regions by the inventors’ address; organisations by the applicants. 

rowOf = a => a ? periods.find(d => d.corpus === corpus && d.actor_id === a.actor_id && d.period === period) : undefined
profileOf = a => {
  const r = rowOf(a)
  if (!r || r.n === 0) return []
  return cats.map(c => ({who: a.actor, category: c.name, k: r[c.category], n: r.n, share: r[c.category] / r.n}))
}
p1 = profileOf(actor)
p2 = (other && actor && other.actor_id !== actor.actor_id) ? profileOf(other) : []
world = profileOf(actors.find(d => d.corpus === corpus && d.actor_id === "A:World"))
// a sentence that sums up the profile
{
  if (!actor) return md`*No actor of this type here: choose another type, or switch off "Only Piedmont".*`
  if (p1.length === 0) return md`**${actor.actor}** has no ${corpus.toLowerCase()} in ${period}.`
  const top = [...p1].sort((a, b) => b.share - a.share)[0]
  const spec = p1.map((d, i) => ({...d, index: d.share / world[i].share}))
                 .filter(d => d.k >= 5).sort((a, b) => b.index - a.index)[0]
  return md`**${actor.actor}** · ${fmt(p1[0].n)} ${corpus.toLowerCase()} (${period}).
  Its largest category is **${top.category}** (${pct(top.share)} of its documents)${spec ?
  `; the one where it is most specialised is **${spec.category}** (${spec.index.toFixed(2)} times the world's share).` : "."}`
}
// 1. The share of the documents in each category: the actor and the comparison
Plot.plot({
  width, height: 460, marginLeft: 200,
  x: {label: "Share of the documents (%) →", percent: true, grid: true},
  y: {axis: null},
  fy: {label: null, domain: cats.map(c => c.name)},
  color: {domain: [actor?.actor, other?.actor], range: ["#c8102e", "#9a9a9a"], legend: true},
  marks: [
    Plot.barX([...p1, ...p2], {x: "share", y: "who", fy: "category", fill: "who",
      title: d => `${d.who}\n${d.category}\n${fmt(d.k)} of ${fmt(d.n)} documents (${pct(d.share)})`, tip: true}),
    Plot.ruleX([0])
  ]
})

A document can be in several categories, so the shares do not add up to 100%.

Specialisation compared with the world

spec = p1.map((d, i) => ({...d, index: d.share / world[i].share, few: d.k < 5})).filter(d => d.k > 0)
actor?.actor_id === "A:World" ? md`*The world is the reference: its index is 1 in every category.*` :
Plot.plot({
  width, height: 300, marginLeft: 200,
  x: {type: "log", label: "Specialisation index (1 = world average, log scale) →", grid: true, tickFormat: "~g"},
  y: {label: null, domain: cats.map(c => c.name)},
  marks: [
    Plot.barX(spec, {y: "category", x1: d => 1, x2: "index",
      fill: d => d.index >= 1 ? "#c8102e" : "#9a9a9a", fillOpacity: d => d.few ? 0.35 : 1,
      title: d => `${d.category}\nindex ${d.index.toFixed(2)} · ${fmt(d.k)} documents${d.few ? " (few)" : ""}`, tip: true}),
    Plot.ruleX([1], {strokeWidth: 1.5})
  ]
})

Over time

trendRows = actor ? trend.filter(d => d.corpus === corpus && d.actor_id === actor.actor_id) : []
trendLong = trendRows.filter(d => d.n >= 10)
  .flatMap(d => cats.map(c => ({window: d.period, category: c.name, k: d[c.category], n: d.n, share: d[c.category] / d.n})))
windows = ["1980-84", "1985-89", "1990-94", "1995-99", "2000-04", "2005-09", "2010-14", "2015-19", "2020-25"]

Plot.plot({
  width, height: 150, marginLeft: 60,
  x: {domain: windows, label: null},
  y: {label: "Documents", grid: true},
  marks: [Plot.barY(trendRows, {x: "period", y: "n", fill: "#54565a", title: d => `${d.period}: ${fmt(d.n)} documents`, tip: true}),
          Plot.ruleY([0])]
})
trendLong.length === 0 ? md`*Too few documents to show a trend.*` :
Plot.plot({
  width, height: 360, marginLeft: 60,
  x: {domain: windows, label: "Five-year window"},
  y: {label: "Share of the documents (%)", percent: true, grid: true},
  color: {domain: cats.map(c => c.name), range: catColours, legend: true},
  marks: [
    Plot.line(trendLong, {x: "window", y: "share", stroke: "category", strokeWidth: 2}),
    Plot.dot(trendLong, {x: "window", y: "share", fill: "category", r: 3,
      title: d => `${d.category}\n${d.window}: ${fmt(d.k)} of ${fmt(d.n)} (${pct(d.share)})`, tip: true})
  ]
})

The last window, 2020–25, has six years.
